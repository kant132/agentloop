# 暴露面扫描设计

Java 代码审计工具的资产存储、缓存与对账方案。用于在代码审计过程中沉淀路由表、调用链、方法体、注解、安全组件(filter/interceptor)等资产,供后续主管和专家复用。

## 1. 目录结构

```
agentloop\项目\
├── 注解_public.json              ← 全局公共注解(跨项目共享)
└── {groupId}\
    ├── 注解\
    │   └── specific.json         ← 项目特有注解(FQN 以 groupId 开头)
    ├── routes.json
    ├── chains\
    │   └── {route_sigHash}.json
    ├── filters.json
    └── interceptors.json
```

项目相关的所有资产(路由、链、filter、interceptor)统一放在 `{groupId}\` 目录下,按项目隔离。

## 2. 文件 Schema

### 2.0 注解_public.json(全局公共注解)

```json
{
  "schema_version": "1.0",
  "annotations": [
    {
      "annotation_type": "routing",
      "fqn": "org.springframework.web.bind.annotation.GetMapping",
      "name": "GetMapping",
      "description": "Spring MVC GET 请求映射注解"
    }
  ]
}
```

说明:
- 全局资产,所有项目共享
- 包含 Spring 标准注解、JDK 注解等通用注解
- 不包含项目自定义注解

### 2.1 {groupId}\注解\specific.json(项目特有注解)

```json
{
  "groupId": "org.owasp.webgoat",
  "annotations": [
    {
      "annotation_type": "auth_check",
      "fqn": "org.owasp.webgoat.container.authorization.WebGoatUserRequired",
      "name": "WebGoatUserRequired",
      "description": "WebGoat 认证校验注解,要求已登录",
      "body": "@Target(ElementType.METHOD)\n@Retention(RetentionPolicy.RUNTIME)\npublic @interface WebGoatUserRequired {\n    String value() default \"\";\n}",
      "body_hash": "a1b2c3d4e5f6g7h8"
    }
  ]
}
```

说明:
- 项目私有资产,按 groupId 隔离
- 包含项目自定义注解(FQN 以 groupId 开头)
- 比 public 多 `body` 和 `body_hash` 字段(用于变更检测)

### 2.2 routes.json(路由完整审计状态表 - 主表)

```json
{
  "groupId": "org.owasp.webgoat",
  "routes": [
    {
      "signature_hash": "9c1a3f4e2b7d8c5a",

      "method_name": "listUsers",
      "file_path": "src/main/java/org/owasp/webgoat/lessons/missingac/MissingFunctionACUsers.java",
      "method_type": "GET",
      "route_path": "/access-control/users",
      "annotation_fqn": "org.springframework.web.bind.annotation.GetMapping",
      "status": "pending",
      "score": 0,
      "audit_analysis": {
        "business_logic": "",
        "taint_analysis": "",
        "sink_detected": ["org.owasp.webgoat.lessons.missingac.UserRepository#findAllUsers"]
      }
    }
  ]
}
```

### 2.3 chains/{route_sigHash}.json(文件名即路由签名 hash)

```json
{
  "groupId": "org.owasp.webgoat",
  "signature_hash": "9c1a3f4e2b7d8c5a",
  "chains": [
    ["m1_hash", "m2_hash", "m3_hash"],
    ["m1_hash", "m4_hash"]
  ],
  "status": "pending",
  "score": 0,
  "audit_analysis": {}
}
```

- 每条调用链是一个有序 hash 数组
- 通过 hash 在 Memurai `{groupId}:method:{sigHash}` 中取完整 body
- 一个路由对应多条调用链(多个分支)

### 2.4 filters.json / interceptors.json

```json
      "status": "pending",

```

## 3. Memurai 缓存

只缓存 method body,TTL 统一 24h。每轮审计启动时 `DEL {groupId}:*`。

### 3.1 Method body 缓存 `{groupId}:method:{sigHash}`

```json
{
  "hashkey": "9c1a3f4e2b7d8c5a",
  "fqn": "org.owasp.webgoat.lessons.missingac.MissingFunctionACUsers#listUsers",
  "body": "public ModelAndView listUsers() {\n    // sink: org.owasp.webgoat.lessons.missingac.UserRepository#findAllUsers\n    List<User> allUsers = userRepository.findAllUsers();\n    return model;\n}"
}
```

字段:
- `hashkey`:签名 hash
- `fqn`:方法全限定名
- `body`:完整方法体,已在第三方调用行前插入 `// sink: <FQN>` 注释

## 5. 数据流

0. 加载注解分类表:
   - 读取 `注解_public.json`(全局)
   - 读取 `{groupId}\注解\specific.json`(项目特有)
   - 构建 annotation_fqn → classification(public/specific)的映射
1. 扫描路由注解 → 查注解分类表 → 写入 `routes.json`
2. 对每个路由构建调用链,方法体生成签名 hash
3. 写入 `chains/{sigHash}.json`,chain 数组只存 hash
4. 所有方法体(已注入 sink 注释)写入 Memurai `{groupId}:method:{sigHash}`
5. 扫描 Filter/Interceptor → 写入对应 `filters.json` / `interceptors.json`

## 6. 分类逻辑

```python
def classify_route(annotation_fqn, global_public, project_specific):
    """
    根据注解 FQN 判断路由分类

    Returns:
        "public": 全局公共注解(如 @GetMapping)
        "specific": 项目特有注解(FQN 以 groupId 开头)
        "unknown": 未分类(需人工判断)
    """
    # 检查是否在公共注解表中
    public_fqns = {a["fqn"] for a in global_public["annotations"]}
    if annotation_fqn in public_fqns:
        return "public"

    # 检查是否在项目特有注解表中
    specific_fqns = {a["fqn"] for a in project_specific["annotations"]}
    if annotation_fqn in specific_fqns:
        return "specific"

    # FQN 以项目 groupId 开头 → 项目特有
    project_group_id = project_specific["groupId"]
    if annotation_fqn.startswith(project_group_id):
        return "specific"

    return "unknown"
```

## 7. 对账(运行时内存校验,不落盘)

```python
def validate(project_root, groupId):
    routes = json.load(open(project_root/groupId/'routes.json'))['routes']
    chain_files = list((project_root/groupId/'chains').glob('*.json'))

    # 路由总数 == chain 文件数
    assert len(routes) == len(chain_files)

    # chain 文件名必须能与 routes.signature_hash 一一对应
    route_hashes = {r['signature_hash'] for r in routes}
    chain_hashes = {f.stem for f in chain_files}
    assert route_hashes == chain_hashes

    # 新增:注解分类表一致性
    public_annotations = json.load(open(project_root/'注解_public.json'))['annotations']
    specific_annotations = json.load(open(project_root/groupId/'注解'/'specific.json'))['annotations']

    # 检查 routes 中的 annotation_fqn 是否都在注解表中
    route_annotation_fqns = {r['annotation_fqn'] for r in routes}
    public_fqns = {a['fqn'] for a in public_annotations}
    specific_fqns = {a['fqn'] for a in specific_annotations}
    all_annotation_fqns = public_fqns | specific_fqns

    missing_fqns = route_annotation_fqns - all_annotation_fqns
    if missing_fqns:
        raise ValueError(f"路由引用了未定义的注解: {missing_fqns}")
```

## 8. 关键函数

```python
def sig_hash(fqn, params):
    """方法签名 hash(16 位 md5)"""
    return md5(f"{fqn}({','.join(params)})")[:16]


def fqn_to_method_name(fqn):
    """从类#方法或包.类.方法 提取方法名"""
    return fqn.rsplit("#", 1)[-1] if "#" in fqn else fqn.rsplit(".", 1)[-1]


def inject_sink_comment(line, third_party_calls, fqn_sink_set):
    """为含第三方 sink 调用的代码行上方插入 '// sink: <FQN>' 注释"""
    if line.strip().startswith('//'):
        return line
    for call in third_party_calls:
        if call in fqn_sink_set and fqn_to_method_name(call) in line:
            indent = len(line) - len(line.lstrip())
            return (' ' * indent) + f'// sink: {call}\n' + line
    return line


def get_body(groupId, sig_hash, memurai):
    """从缓存取方法体"""
    rec = json.loads(memurai.get(f"{groupId}:method:{sig_hash}") or "null")
    return rec['body'] if rec else None


def detect_annotation_changes(project_dir, groupId):
    """检测 specific.json 中注解的变更(body_hash 校验)"""
    specific_path = project_dir/groupId/'注解'/'specific.json'
    specific_data = json.load(open(specific_path))

    changes = []
    for a in specific_data["annotations"]:
        current_hash = md5(a["body"])[:16]
        recorded_hash = a.get("body_hash")
        if recorded_hash and current_hash != recorded_hash:
            changes.append({
                "fqn": a["fqn"],
                "old_hash": recorded_hash,
                "new_hash": current_hash
            })

    return changes
```

## 9. 设计约束

- `body` 字段只在 Memurai 中存在一处,不落地任何文件
- 所有文件、缓存都通过 `signature_hash` 或 `fqn` 串联引用
- 审计轮次启动时 `DEL {groupId}:*` 全清,重新构建
- `注解_public.json` 是全局资产,跨项目共享,不按 groupId 隔离
- `specific.json` 按 groupId 隔离,包含项目自定义注解
